تعديل إحصائيات الإحالة

الحقل الفارغ يعيد هذا البند للحساب التلقائي. إدخال 0 يعرض صفرًا. «صالح» في الحساب التلقائي الأصلي هو عدد عمليات إيداع الفريق المعتمدة، وليس عدد الأعضاء الفريدين.

هذه تعديلات لإحصائيات قسم الفريق فقط؛ لا تغيّر رصيد محفظة المستخدم أو سجلات الإيداع والسحب أو آلية إضافة العمولة التلقائية القائمة. تعديل العمولة هنا لا يضيف مالًا للمحفظة. عدد المستوى الأول يتبع حجم الفريق ما لم تضع له قيمة مستقلة.

تعليمات التركيب والحماية — ضرورية قبل الاستخدام

ارفع index.html وadmin.html في المجلد نفسه على استضافة HTTP/HTTPS. سجّل دخول حسابك، وأضف إلى مستنده users/{UID} الحقل admin من نوع Boolean بالقيمة true من Firebase Console أو خادم موثوق فقط. القيمة false أو النص "true" أو غياب الحقل يمنع الدخول. لا تمنح المستخدمين حق تعديل هذا الحقل.

تُحفظ التعديلات في teamStats/{UID} وتُعرض للمستخدم مباشرة عند الاتصال. الحقول الفارغة تحفظ كـ null، ويمكن الرجوع للحساب التلقائي دون المساس بالعمليات المالية. تتطلب معاينة القيم التلقائية أن تسمح قواعدك للأدمن بقراءة الإحالات وطلبات الإيداع والسحب الحالية.

حماية واجهة HTML ليست حماية لقاعدة البيانات. القواعد المنشورة حاليًا غير مرفقة ولم يتم تعديلها أو نشرها. ادمج المقطع أدناه داخل قواعد Firestore الحالية، ولا تستبدل قواعد المشروع به. يجب تقييد أي قاعدة عامة متداخلة تسمح بالكتابة، وإضافة منع تعديل admin إلى كل مسار كتابة غير إداري في users؛ قواعد السماح المتداخلة تُجمع بمنطق OR ولا يمكن إلغاؤها بقاعدة منع أخرى.

المقطع يضيف صلاحيات قراءة المستخدمين للأدمن فقط، ويقصر تعديل إحصائيات الفريق على الأدمن، ويسمح لكل مستخدم بقراءة إحصائياته. حافظ على شروط الأمان الحالية للتسجيل والمكافآت والمحافظ، وأضف لها قيود حماية حقل admin الموضحة في نهاية المقطع. اختبر بحساب غير أدمن أنه لا يستطيع إنشاء admin=true أو تغييره، ولا الكتابة إلى teamStats مباشرة.

// Merge INSIDE the existing match /databases/{database}/documents block.
// This is an integration patch, NOT a replacement for your existing rules.
function marexTeamAdmin() {
  return request.auth != null
    && exists(/databases/$(database)/documents/users/$(request.auth.uid))
    && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('admin', false) == true;
}
function marexTeamCount(v) {
  return v == null || (v is int && v >= 0 && v <= 999999999999);
}
function marexTeamMoney(v) {
  return v == null || (v is number && v >= 0 && v <= 999999999999);
}
function marexTeamStatsValid(uid) {
  let d = request.resource.data;
  return d.keys().hasOnly([
      'uid','teamCount','teamDeposits','teamWithdrawals','teamCommission',
      'level1Count','level1Valid','level2Count','level2Valid','level3Count','level3Valid',
      'revision','updatedBy','updatedAt'])
    && d.uid == uid
    && exists(/databases/$(database)/documents/users/$(uid))
    && d.updatedBy == request.auth.uid
    && d.updatedAt == request.time
    && d.revision is int
    && marexTeamCount(d.teamCount)
    && marexTeamMoney(d.teamDeposits)
    && marexTeamMoney(d.teamWithdrawals)
    && marexTeamMoney(d.teamCommission)
    && marexTeamCount(d.level1Count) && marexTeamCount(d.level1Valid)
    && marexTeamCount(d.level2Count) && marexTeamCount(d.level2Valid)
    && marexTeamCount(d.level3Count) && marexTeamCount(d.level3Valid);
}
match /teamStats/{uid} {
  allow get: if request.auth != null
    && (request.auth.uid == uid || marexTeamAdmin());
  allow list: if marexTeamAdmin();
  allow create: if marexTeamAdmin() && marexTeamStatsValid(uid)
    && request.resource.data.revision == 1;
  allow update: if marexTeamAdmin() && marexTeamStatsValid(uid)
    && request.resource.data.revision == resource.data.get('revision', 0) + 1;
  allow delete: if marexTeamAdmin();
}
// Add these admin read permissions to the existing users match.
match /users/{uid} {
  allow get, list: if marexTeamAdmin();
}
